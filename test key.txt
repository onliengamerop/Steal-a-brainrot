local function CopyToClipboard(text)
    local fn = setclipboard or toclipboard or (syn and syn.write_clipboard) or (Clipboard and Clipboard.set)
    if type(fn) ~= "function" then
        return false
    end
    return (pcall(fn, text))
end

local function CleanKey(raw)
    return (tostring(raw or ""):gsub("[^%w]", ""))
end

local function SaveKey(key)
    if type(writefile) == "function" then
        pcall(writefile, Saved_Key_Path, key)
    end
end

local function ClearSavedKey()
    pcall(function()
        if type(isfile) == "function" and isfile(Saved_Key_Path) then
            if type(delfile) == "function" then
                delfile(Saved_Key_Path)
            else
                writefile(Saved_Key_Path, "")
            end
        end
    end)
end

local function ReadSavedKey()
    local key
    pcall(function()
        if type(isfile) == "function" and isfile(Saved_Key_Path) then
            key = CleanKey(readfile(Saved_Key_Path))
        end
    end)
    if key and key ~= "" then
        return key
    end
    return nil
end

end

local function LoadAPI()
    if API then
        return true
    end
    local ok, result = pcall(function()
        local source = game:HttpGet(Luarmor_URL)
        local factory, compileError = loadstring(source)
        assert(type(factory) == "function", tostring(compileError or "Could not compile the Luarmor library."))
        local library = factory()
        assert(type(library) == "table", "Luarmor returned an invalid library.")
        assert(type(library.check_key) == "function", "Luarmor check_key is unavailable.")
        library.script_id = Script_ID
        return library
    end)
    if ok then
        API = result
        return true
    end
    return false, tostring(result)
end

local function RequestKeyStatus(key)
    if #Script_ID ~= 32 or not Script_ID:match("^%x+$") then
        return { code = "SCRIPT_ID_INVALID", message = StatusMessages.SCRIPT_ID_INVALID }
    end

    local lastError = "No response from Luarmor."
    for attempt = 1, 3 do
        local loaded, loadError = LoadAPI()
        if loaded then
            local ok, status = pcall(API.check_key, key)
            if ok and type(status) == "table" and type(status.code) == "string" then
                if not RetryOn[status.code] or attempt == 3 then
                    return status
                end
                lastError = tostring(status.message or status.code)
            elseif ok then
                lastError = "Luarmor returned an incomplete response."
            else
                lastError = tostring(status)
            end
        else
            lastError = loadError
        end
        if attempt < 3 then
            task.wait(attempt)
        end
    end
    return nil, lastError
end

local function RunHub(key)
    SetGlobalKey(key)
    local ok, err
    if API and type(API.load_script) == "function" then
        ok, err = pcall(API.load_script)
    else
        ok, err = pcall(function()
            loadstring(game:HttpGet(Loader_URL))()
        end)
    end
    print(("[Ajjan KeyCheck] RunHub pcall ok=%s err=%s"):format(tostring(ok), tostring(err)))
    if not ok then
        ReleaseRunLock()
        warn("Ajjan Hub loader failed: " .. tostring(err))
    end
end

-- Diagnostic-only recheck: fires an independent, uncached check_key call
-- immediately before we hand the key to the loader. This does NOT assume
-- what's wrong yet -- it just tells us, on the next failure, whether
-- Luarmor's own key-check API still agrees the key is valid a few hundred
-- ms later. If it does and the loader still shows "Whitelist Error", the
-- bug is inside the loader's own gate (HWID/session/script-id), not in
-- our check_key logic. If it doesn't, we've caught a real state flip
-- (rate limit, HWID race, blacklist) and can stop before ever touching
-- the loader.
local function RecheckBeforeLoad(key)
    local status, networkError = RequestKeyStatus(key)
    if status then
        print(("[Ajjan KeyCheck] pre-load recheck code=%s message=%s"):format(
            tostring(status.code), tostring(status.message)))
    else
        print(("[Ajjan KeyCheck] pre-load recheck network error: %s"):format(tostring(networkError)))
    end
    return status
end

local function CheckLuarmorKey(key)
    if isCheckingKey or verified then
        return
    end
    SetBusy(true)
    SetStatus("Checking key...", Yellow)

    local status, networkError = RequestKeyStatus(key)

    if status and status.code == "KEY_VALID" then
        SetStatus("Confirming before load...", Yellow)
        local recheck = RecheckBeforeLoad(key)

        if not recheck or recheck.code ~= "KEY_VALID" then
            -- The first check said valid, the recheck disagreed a moment
            -- later. Don't push into the loader at all -- surface the
            -- real second-check result so we know what actually flipped.
            local recheckMessage = recheck
                and (StatusMessages[recheck.code] or ("Key check failed: " .. tostring(recheck.message or recheck.code)))
                or "Could not confirm key before loading (no response from Luarmor)."
            Notify("❌ Pre-load check failed: " .. recheckMessage, Red)
            SetStatus("Recheck failed — retry", Red)
            SetBusy(false)
            return
        end

        verified = true
        SaveKey(key)
        SetGlobalKey(key)
        SetStatus("Verified — loading...", Green)
        Notify("Key verified! Loading hub...", Green)
        task.wait(0.6)
        if ScreenGui.Parent then
            ScreenGui:Destroy()
        end
        RunHub(key)
        return
    end

    local message
    if status then
        message = StatusMessages[status.code] or ("Key check failed: " .. tostring(status.message or status.code))
        if ClearSavedOn[status.code] then
            ClearSavedKey()
        end
    else
        message = "Could not reach Luarmor: " .. tostring(networkError)
    end

    Notify("❌ " .. message, Red)
    if status and status.code == "KEY_HWID_LOCKED" then
        SetStatus("HWID locked", Orange)
    else
        SetStatus("Key not verified — retry", Red)
    end
    SetBusy(false)
end

local function SubmitKey(raw)
    local key = CleanKey(raw)
    if key == "" then
        Notify("⚠️ Paste your key first.", Orange)
        return
    end
    if #key ~= 32 then
        Notify("⚠️ Keys are 32 characters. Yours is " .. #key .. ".", Orange)
        return
    end
    TextBox.Text = key
    task.spawn(CheckLuarmorKey, key)
end
